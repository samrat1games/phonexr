cmake_minimum_required(VERSION 3.21)
project(PhoneXRDesktop LANGUAGES CXX)
option(PHONEXR_LITE "Build PhoneXR Lite Share" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network Gui)
add_subdirectory(../steamvr ${CMAKE_BINARY_DIR}/steamvr)
qt_add_executable(phonexr-share main.cpp resources.qrc)
if(PHONEXR_LITE)
  target_compile_definitions(phonexr-share PRIVATE PHONEXR_LITE=1)
  set_target_properties(phonexr-share PROPERTIES OUTPUT_NAME "phonexr-lite-share")
endif()
target_link_libraries(phonexr-share PRIVATE Qt6::Widgets Qt6::Network Qt6::Gui)
install(TARGETS phonexr-share RUNTIME DESTINATION bin)
install(DIRECTORY ${CMAKE_BINARY_DIR}/phonexr DESTINATION share/phonexr-share/steamvr)
